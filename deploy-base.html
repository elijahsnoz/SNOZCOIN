// SPDX-License-Identifier: MIT pragma solidity ^0.8.20; /** * @title SNOZCOIN User Profiles * @notice Stores user profile data on-chain for the SNOZCOIN platform * @dev Deployed on Base Sepolia Testnet */ contract SnozcoinProfiles { struct Profile { string username; string country; uint8 userType; string avatarSeed; string ipfsHash; uint256 createdAt; uint256 updatedAt; bool isVerified; bool exists; } address public owner; uint256 public totalProfiles; bool public paused; mapping(address => Profile) public profiles; mapping(string => address) public usernameToAddress; mapping(address => bool) public admins; event ProfileCreated(address indexed wallet, string username, uint8 userType, uint256 timestamp); event ProfileUpdated(address indexed wallet, uint256 timestamp); event UserVerified(address indexed wallet, uint256 timestamp); modifier onlyOwner() { require(msg.sender == owner, "Not authorized"); _; } modifier onlyAdmin() { require(msg.sender == owner || admins[msg.sender], "Not authorized"); _; } modifier whenNotPaused() { require(!paused, "Contract is paused"); _; } constructor() { owner = msg.sender; admins[msg.sender] = true; } function registerProfile( string calldata username, string calldata country, uint8 userType, string calldata avatarSeed ) external whenNotPaused { require(!profiles[msg.sender].exists, "Profile exists"); bytes memory ub = bytes(username); require(ub.length >= 3 && ub.length <= 20, "Username: 3-20 chars"); require(usernameToAddress[username] == address(0), "Username taken"); require(bytes(country).length == 3, "Country: 3 chars"); require(userType >= 1 && userType <= 4, "UserType: 1-4"); profiles[msg.sender] = Profile({ username: username, country: country, userType: userType, avatarSeed: avatarSeed, ipfsHash: "", createdAt: block.timestamp, updatedAt: block.timestamp, isVerified: false, exists: true }); usernameToAddress[username] = msg.sender; totalProfiles++; emit ProfileCreated(msg.sender, username, userType, block.timestamp); } function updateProfile(string calldata country, uint8 userType, string calldata avatarSeed) external whenNotPaused { require(profiles[msg.sender].exists, "No profile"); profiles[msg.sender].country = country; profiles[msg.sender].userType = userType; profiles[msg.sender].avatarSeed = avatarSeed; profiles[msg.sender].updatedAt = block.timestamp; emit ProfileUpdated(msg.sender, block.timestamp); } function getProfile(address wallet) external view returns ( string memory, string memory, uint8, string memory, string memory, uint256, uint256, bool ) { Profile storage p = profiles[wallet]; require(p.exists, "No profile"); return (p.username, p.country, p.userType, p.avatarSeed, p.ipfsHash, p.createdAt, p.updatedAt, p.isVerified); } function hasProfile(address wallet) external view returns (bool) { return profiles[wallet].exists; } function isUsernameAvailable(string calldata u) external view returns (bool) { return usernameToAddress[u] == address(0); } function verifyUser(address w) external onlyAdmin { profiles[w].isVerified = true; emit UserVerified(w, block.timestamp); } function pause() external onlyOwner { paused = true; } function unpause() external onlyOwner { paused = false; } }
